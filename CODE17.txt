from itertools import permutations

# Read string S and integer k from input
s, k = input().split()
k = int(k)

# Sort the string first to ensure lexicographic order
sorted_s = sorted(s)

# Generate permutations of size k
perms = permutations(sorted_s, k)

# Join the characters in each tuple and print on new lines
for p in perms:
    print("".join(p))
