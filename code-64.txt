import sys

if __name__ == "__main__":
   
    n, m = map(int, sys.stdin.readline().split())
   
    arr = []
    for _ in range(n):
        arr.append(list(map(int, sys.stdin.readline().split())))
  
    k = int(sys.stdin.readline().strip())
    

    arr.sort(key=lambda x: x[k])
 
    for row in arr:
        print(*(row))
